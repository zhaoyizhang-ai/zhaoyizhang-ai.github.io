第9讲作业

4.47

A. 用指针引用数组元素的 C 代码如下：

#include <stdio.h>

void bubble_p(long *data, long count)
{
    if (count <= 1)
        return;

    for (long *last = data + count - 1; last > data; --last) {
        for (long *p = data; p < last; ++p) {
            if (*(p + 1) < *p) {
                long t = *(p + 1);
                *(p + 1) = *p;
                *p = t;
            }
        }
    }
}

int main(void)
{
    long data[] = {5, -1, 3, 0, -7, 3, 2};
    long count = sizeof(data) / sizeof(data[0]);

    bubble_p(data, count);
    for (long *p = data; p < data + count; ++p)
        printf("%ld ", *p);
    printf("\n");
    return 0;
}

last 指向本轮未排序部分的最后一个元素，p 从数组起点向后移动，每次比较 *p 与 *(p + 1)，必要时交换。一轮结束后，最大值被移到 last，下一轮将 last 向前移动一个元素。count ≤ 1 时直接返回。

测试数组为 {5, -1, 3, 0, -7, 3, 2}，运行结果为：

-7 -1 0 2 3 3 5

B. 对应的 Y86-64 程序及测试代码如下：

# Test program
.pos 0
    irmovq stack, %rsp
    irmovq data, %rdi
    irmovq $7, %rsi
    call bubble_p
    halt

# %rdi = data, %rsi = count
bubble_p:
    andq %rsi, %rsi
    jle Done
    irmovq $1, %r8
    subq %r8, %rsi
    je Done
    addq %rsi, %rsi
    addq %rsi, %rsi
    addq %rsi, %rsi
    addq %rdi, %rsi       # last = data + 8 * (count - 1)
    irmovq $8, %r8

Outer:
    rrmovq %rdi, %rcx    # p = data
Inner:
    mrmovq 0(%rcx), %rax # a = *p
    mrmovq 8(%rcx), %rdx # b = *(p + 1)
    rrmovq %rax, %r9
    subq %rdx, %r9       # signed comparison: a - b
    jle NoSwap
    rmmovq %rdx, 0(%rcx)
    rmmovq %rax, 8(%rcx)
NoSwap:
    addq %r8, %rcx       # ++p
    rrmovq %rcx, %r9
    subq %rsi, %r9
    jl Inner            # p < last
    subq %r8, %rsi       # --last
    rrmovq %rsi, %r9
    subq %rdi, %r9
    jg Outer            # last > data
Done:
    ret

.pos 0x400
data:
    .quad 5
    .quad -1
    .quad 3
    .quad 0
    .quad -7
    .quad 3
    .quad 2

.pos 0x1000
stack:

%rdi 保存数组起始地址，%rsi 最初保存 count，随后改为保存 last；%rcx 保存 p，%r8 保存步长 8。将 count - 1 连续加倍三次，再加上数组起始地址，就得到最后一个元素的地址。

subq %rdx, %r9 根据 a - b 设置条件码，jle 在 a ≤ b 时跳过交换。内外层循环分别以 p < last 和 last > data 为继续条件。用 yas 汇编后在 yis 上运行，程序正常以 HLT 状态结束，数组内容变为 {-7, -1, 0, 2, 3, 3, 5}，与 C 程序一致。

4.48

将 4.47 中 Inner 下读取、比较和交换元素的代码替换为以下部分，保留其后的循环控制代码：

    mrmovq 0(%rcx), %rax # a = *p
    mrmovq 8(%rcx), %rdx # b = *(p + 1)
    rrmovq %rax, %r10    # save a
    rrmovq %rax, %r9
    subq %rdx, %r9       # a - b
    cmovg %rdx, %rax     # if a > b, left = b
    cmovg %r10, %rdx     # if a > b, right = a
    rmmovq %rax, 0(%rcx)
    rmmovq %rdx, 8(%rcx)

%r10 先保存原来的 a。subq 根据 a - b 设置条件码，当 a > b 时，两条 cmovg 分别将 b 送入 %rax、将原来的 a 送入 %rdx，实现交换；否则两者保持原值。最后将结果写回数组。

cmovg 不修改条件码，因此两条条件传送使用的是同一次比较的结果。这段比较交换代码没有跳转，只用了 2 条条件传送，满足最多使用 3 次的要求。

4.49

比较交换部分改写如下：

    mrmovq 0(%rcx), %rax # a = *p
    mrmovq 8(%rcx), %rdx # b = *(p + 1)
    rrmovq %rax, %r10
    xorq %rdx, %r10      # d = a ^ b
    rrmovq %rax, %r9
    subq %rdx, %r9       # a - b
    irmovq $0, %r11      # irmovq preserves condition codes
    cmovg %r10, %r11     # mask = (a > b) ? d : 0
    xorq %r11, %rax
    xorq %r11, %rdx
    rmmovq %rax, 0(%rcx)
    rmmovq %rdx, 8(%rcx)

先在 %r10 中计算 d = a ^ b，再通过 a - b 设置条件码。%r11 初始为 0，irmovq 不修改条件码；当 a > b 时，唯一一条 cmovg 将 d 送入 %r11，否则 %r11 仍为 0。

当 a ≤ b 时，与 0 异或不会改变 a、b；当 a > b 时：

a⊕(a⊕b)=b,b⊕(a⊕b)=a.a \oplus (a \oplus b) = b,\qquad b \oplus (a \oplus b) = a.a⊕(a⊕b)=b,b⊕(a⊕b)=a.

因此，两条 xorq 完成交换，再将结果写回数组。整段比较交换代码没有跳转，恰好使用 1 条条件传送。

4.48、4.49 的程序均经过 yis 测试，排序结果与 4.47 一致；空数组、单元素、重复值、逆序及含正负数的情况也均通过测试。
